from __future__ import annotations

from copy import deepcopy
from typing import Any, Dict, List, Optional, Tuple, Type, Union

import numpy as np
import torch
from torch.utils.data import Dataset

from olmo.exceptions import OLMoEnvironmentError

from ..aliases import PathOrStr
from ..config import InstanceFilterConfig
from ..util import _get_s3_client, file_size, get_bytes_range
from .util import find_periodic_sequences, get_document_lengths

class MemMapDataset(Dataset[Dict[str, Any]]):
    """
    A PyTorch :class:`~torch.utils.data.Dataset` backed by one or more numpy memory-mapped arrays
    of token IDs. Token IDs are chunked together into contiguous blocks of ``chunk_size``
    to create instances.

    If the length of a memory-mapped array is not a multiple of ``chunk_size`` the
    remainder of the tokens will be ignored.

    No special tokens are added to the input IDs so it's assumed that if you want
    EOS tokens between documents, for example, those will already be in the memory-mapped array.

    :param paths: Paths to memory-mapped token arrays.
    :param chunk_size: The number of tokens to chunk together into a single instance.
        Generally this should correspond to your model's maximum input length.
    :param memmap_dtype: The numpy datatype of the memory-mapped array.
    :param metadata: Metadata to add to each item. This should be a dictionary or a list of dictionaries
        with the same number of items as there are paths.
    :param include_instance_metadata: If ``True`` (the default), each instance returned from `__getitem__` will
        include the metadata from its source.
    :param generate_attention_mask: If ``True``, each instance returned from ``__getitem__`` will include an
        attention mask generated by masking each padding token.
    :param pad_token_id: The ID of the padding token. Required if ``generate_attention_mask`` is ``True``.
    :param label_mask_paths: Optional paths to ``np.bool_`` memory-mapped arrays of label masks.
    """

    def __init__(
        self,
        *paths: PathOrStr,
        chunk_size: int = 1024,
        memmap_dtype: Union[Type[np.uint8], Type[np.uint16], Type[np.uint32], Type[np.uint64]] = np.uint16,
        metadata: Optional[Union[List[Dict[str, Any]], Dict[str, Any]]] = None,
        include_instance_metadata: bool = True,
        generate_attention_mask: bool = False,
        generate_doc_lengths: bool = False,
        pad_token_id: Optional[int] = None,
        eos_token_id: Optional[int] = None,
        label_mask_paths: Optional[List[PathOrStr]] = None,
        instance_filter_config: Optional[InstanceFilterConfig] = None,
    ):
        if not paths:
            raise ValueError("At least one path is required")

        if generate_attention_mask and pad_token_id is None:
            raise ValueError("'pad_token_id' is required for 'generate_attention_mask'")

        if generate_doc_lengths and eos_token_id is None:
            raise ValueError("'eos_token_id' is required for 'generate_cu_doc_lengths'")

        if label_mask_paths and len(label_mask_paths) != len(paths):
            raise ValueError("There must be the same number of 'label_mask_paths' as there are 'paths'")

        if isinstance(metadata, list):
            if len(metadata) != len(paths):
                raise ValueError("'metadata' should have the same length as the number of file paths")
        else:
            metadata = [metadata or {}] * len(paths)

        self._memmap_paths = paths
        self._metadata = metadata
        self._label_mask_paths = label_mask_paths
        self._chunk_size = chunk_size
        self._mmap_offsets: Optional[List[Tuple[int, int]]] = None
        self._num_instances: Optional[int] = None
        self.dtype = memmap_dtype
        self._include_instance_metadata = include_instance_metadata
        self._generate_attention_mask = generate_attention_mask
        self._generate_doc_lengths = generate_doc_lengths
        self._pad_token_id = pad_token_id
        self._eos_token_id = eos_token_id
        self.instance_filter_config = instance_filter_config

    @property
    def chunk_size(self) -> int:
        return self._chunk_size

    @property
    def max_seq_len(self) -> int:
        # For compatibility with composer's SpeedMonitor callback.
        return self.chunk_size

    @property
    def offsets(self) -> List[Tuple[int, int]]:
        # Create the global S3 client up front to work around a threading issue in boto.
        _get_s3_client("s3")
        try:
            _get_s3_client("r2")
        except OLMoEnvironmentError:
            # R2 might not be needed, so ignore this error. We will get an error
            # later if R2 is needed.
            pass
        try:
            _get_s3_client("weka")
        except OLMoEnvironmentError:
            # Weka might not be needed, so ignore this error. We will get an error
            # later if Weka is needed.
            pass

        if self._mmap_offsets is None:
            import concurrent.futures

            self._mmap_offsets = []

            path_to_length: Dict[PathOrStr, int] = {}
            path_to_mask_path: Dict[PathOrStr, PathOrStr] = {}
            mask_path_to_length: Dict[PathOrStr, int] = {}

            with concurrent.futures.ThreadPoolExecutor() as executor:
                path_futures = []
                mask_path_futures = []
                for i, path in enumerate(self._memmap_paths):
                    path_futures.append(executor.submit(self._get_file_length, path))
                    if self._label_mask_paths is not None:
                        mask_path = self._label_mask_paths[i]
                        path_to_mask_path[path] = mask_path
                        mask_path_futures.append(executor.submit(self._get_file_length, mask_path, np.bool_))

                for future in concurrent.futures.as_completed(path_futures):
                    path, length = future.result()
                    path_to_length[path] = length

                for future in concurrent.futures.as_completed(mask_path_futures):
                    path, length = future.result()
                    mask_path_to_length[path] = length

            start_offset = 0
            for path in self._memmap_paths:
                length = path_to_length[path]
                if mask_path_to_length:
                    mask_path = path_to_mask_path[path]
                    if length != mask_path_to_length[mask_path]:
                        raise ValueError(f"masking file '{mask_path}' should be the same size as '{path}'")
                end_offset = start_offset + length
                self._mmap_offsets.append((start_offset, end_offset))
                start_offset += length
        return self._mmap_offsets

    def _read_chunk_from_memmap(self, path: PathOrStr, index: int, dtype=None) -> torch.Tensor:
        dtype = dtype or self.dtype
        item_size = dtype(0).itemsize
        bytes_start = index * item_size * self._chunk_size
        num_bytes = item_size * self._chunk_size
        buffer = get_bytes_range(path, bytes_start, num_bytes)
        array = np.frombuffer(buffer, dtype=dtype)
        if dtype == np.bool_:
            return torch.tensor(array)
        else:
            return torch.tensor(array.astype(np.int_), dtype=torch.long)

    def _get_file_length(self, path, dtype=None) -> Tuple[PathOrStr, int]:
        dtype = dtype or self.dtype
        item_size = dtype(0).itemsize
        return path, file_size(path) // (item_size * self._chunk_size)

    def __len__(self) -> int:
        if self._num_instances is None:
            self._num_instances = self.offsets[-1][1]
        return self._num_instances

    def __getitem__(self, index: int) -> Dict[str, Any]:
        index = int(index)  # in case this is a numpy int type.
        pos_index = index if index >= 0 else len(self) + index

        # The index of the memmap array within 'self.memmaps'
        memmap_index: Optional[int] = None
        # The 'index' relative to the corresponding memmap array.
        memmap_local_index: Optional[int] = None
        for i, (offset_start, offset_end) in enumerate(self.offsets):
            if offset_start <= pos_index < offset_end:
                memmap_index = i
                memmap_local_index = pos_index - offset_start

        if memmap_index is None or memmap_local_index is None:
            raise IndexError(f"{index} is out of bounds for dataset of size {len(self)}")

        # Read the data from file.
        input_ids = self._read_chunk_from_memmap(self._memmap_paths[memmap_index], memmap_local_index)
        out: Dict[str, Any] = {"input_ids": input_ids}
        if self.instance_filter_config is not None:
            out["instance_mask"] = self._validate_instance(input_ids)

        if self._label_mask_paths is not None:
            label_mask = self._read_chunk_from_memmap(
                self._label_mask_paths[memmap_index], memmap_local_index, dtype=np.bool_
            )
            out["label_mask"] = label_mask

        if self._include_instance_metadata:
            metadata = self._metadata[memmap_index]
            out["metadata"] = deepcopy(metadata)

        if self._generate_attention_mask:
            assert self._pad_token_id is not None
            attn_mask = torch.ones_like(input_ids)
            attn_mask.masked_fill_(input_ids == self._pad_token_id, 0)
            out["attention_mask"] = attn_mask

        if self._generate_doc_lengths:
            assert self._eos_token_id is not None
            out["doc_lens"] = get_document_lengths(input_ids, self._eos_token_id)

        return out

    def __add__(self, other: MemMapDataset) -> MemMapDataset:
        """
        Concatenate one :class:`MemMapDataset` with another.
        """
        if not isinstance(other, MemMapDataset):
            raise NotImplementedError(f"Expected another MemMapDataset but got {type(other)}")
        return MemMapDataset(
            *(self._memmap_paths + other._memmap_paths),
            chunk_size=self._chunk_size,
            memmap_dtype=self.dtype,
            metadata=self._metadata + other._metadata,
        )

    def _validate_instance(self, input_ids: torch.Tensor) -> bool:
        # Check for too many repeated ngrams.
        # TODO: update `max_period` per Luca's suggestion.
        if self.instance_filter_config is not None:
            for m in find_periodic_sequences(
                input_ids.numpy(),
                max_period=self.instance_filter_config.repetition_max_period,
                min_period=self.instance_filter_config.repetition_min_period,
            ):
                if m.times >= self.instance_filter_config.repetition_max_count:
                    return False
        return True


def build_memmap_dataset(
    train_config: TrainConfig, data_config: DataConfig, include_instance_metadata: bool = True
) -> MemMapDataset:
    paths: List[str]
    metadata: List[Dict[str, Any]] = []
    if data_config.paths:
        if data_config.datasets:
            raise OLMoConfigurationError("DataConfig.paths is mutually exclusive with DataConfig.datasets")
        paths = data_config.paths
        for path in paths:
            metadata.append({"path": str(path)})
    elif data_config.datasets:
        paths = []
        for label in sorted(data_config.datasets.keys()):
            label_paths = data_config.datasets[label]
            paths.extend(label_paths)
            metadata.extend([{"label": label}] * len(label_paths))
    else:
        raise OLMoConfigurationError("One of DataConfig.paths or DataConfig.datasets is required")
    return MemMapDataset(
        *paths,
        chunk_size=train_config.model.max_sequence_length,
        memmap_dtype=data_config.effective_memmap_dtype,
        metadata=metadata,
        include_instance_metadata=include_instance_metadata,
        pad_token_id=train_config.model.pad_token_id,
        eos_token_id=train_config.model.eos_token_id,
        generate_attention_mask=data_config.generate_attention_mask,
        generate_doc_lengths=data_config.generate_doc_lengths,
        label_mask_paths=cast(Optional[List[PathOrStr]], data_config.label_mask_paths),
        instance_filter_config=data_config.instance_filter,
    )

class IterableDataset(torch.utils.data.IterableDataset[Dict[str, Any]]):
    """
    Adapted from PyTorch's DistributedSampler, this wraps a Dataset or arbitrary sequence
    as an IterableDataset that can be deterministically restarted at any point by setting `start_index`,
    which should be a multiple of your global batch size.
    Similarly `max_examples`, if set, should be a multiple of global batch size.
    """

    def __init__(
        self,
        dataset: Union[Sequence[List[int]], Sequence[torch.Tensor], Sequence[Dict[str, Any]]],
        global_batch_size: int,
        *,
        seed: int = 0,
        epoch: int = 0,
        start_index: int = 0,
        max_examples: Optional[int] = None,
        shuffle: bool = True,
        drop_last: bool = False,
        world_size: Optional[int] = None,
        rank: Optional[int] = None,
        fs_local_rank: Optional[int] = None,
        work_dir: Optional[PathOrStr] = None,
        num_threads: Optional[int] = None,
    ):
        self.dataset = dataset
        self.seed = seed
        self.epoch = epoch
        self.start_index = start_index
        self.max_examples = max_examples
        self.shuffle = shuffle
        self.drop_last = drop_last
        self.rank = rank if rank is not None else get_global_rank()
        self.fs_local_rank = fs_local_rank if fs_local_rank is not None else get_fs_local_rank()
        self.world_size = world_size if world_size is not None else get_world_size()
        # If the dataset length is evenly divisible by # of replicas, then there
        # is no need to drop any data, since the dataset will be split equally.
        if self.drop_last and len(self.dataset) % self.world_size != 0:  # type: ignore[arg-type]
            # Split to nearest available length that is evenly divisible by world size.
            # This is to ensure each rank receives the same amount of data.
            num_samples = math.ceil(
                (len(self.dataset) - self.world_size) / self.world_size  # type: ignore[arg-type]
            )
        else:
            num_samples = math.ceil(len(self.dataset) / self.world_size)  # type: ignore[arg-type]
        self.total_size = num_samples * self.world_size
        self.num_threads = num_threads
        assert global_batch_size % self.world_size == 0
        self.device_batch_size = global_batch_size // self.world_size
        self.global_indices_file: Optional[Path] = None
        self.work_dir = work_dir

        if work_dir is not None:
            self._build_and_save_global_indices()

    def _build_and_save_global_indices(self):
        assert self.work_dir is not None
        self.global_indices_file = Path(self.work_dir) / "global_indices.npy"
        if self.fs_local_rank == 0:
            log.info("Saving global data order indices...")
            self.global_indices_file.parent.mkdir(parents=True, exist_ok=True)
            global_indices = self._build_global_indices()
            global_indices_mmap = np.memmap(
                self.global_indices_file, dtype=np.uint32, mode="w+", shape=(len(global_indices),)
            )
            global_indices_mmap[:] = global_indices
            global_indices_mmap.flush()
            del global_indices_mmap
            log.info("Global data order indices saved to '%s'", self.global_indices_file)
        barrier()

    def _build_global_indices(self) -> np.ndarray:
        assert len(self.dataset) < np.iinfo(np.uint32).max
        indices = np.arange(len(self.dataset), dtype=np.uint32)
        if self.shuffle:
            # Deterministically shuffle based on epoch and seed
            # Torch built-in randomness is not very random, so we use numpy.
            rng = np.random.Generator(np.random.PCG64(seed=self.seed + self.epoch))
            rng.shuffle(indices)

        if not self.drop_last:
            # Add extra samples to make it evenly divisible
            padding_size = self.total_size - len(indices)
            arrays_to_concatenate = [indices]
            while padding_size > 0:
                array_to_concatenate = indices[: min(padding_size, len(indices))]
                arrays_to_concatenate.append(array_to_concatenate)
                padding_size -= len(array_to_concatenate)
                del array_to_concatenate
            indices = np.concatenate(arrays_to_concatenate)
        else:
            # Remove tail of data to make it evenly divisible.
            indices = indices[: self.total_size]
        assert len(indices) == self.total_size
        return indices

    def get_global_indices(self) -> np.ndarray:
        if self.global_indices_file is not None:
            return np.memmap(self.global_indices_file, mode="r", dtype=np.uint32)  # type: ignore
        else:
            return self._build_global_indices()

    def reshuffle(self, epoch: int):
        self.epoch = epoch
        if self.work_dir is not None:
            self._build_and_save_global_indices()

    def __iter__(self) -> Iterator[Dict[str, Any]]:
        indices = self.get_global_indices()

        # Truncate to max_examples.
        if self.max_examples is not None:
            assert self.max_examples % self.world_size == 0
            indices = indices[: self.max_examples]

        # Start at the specified index.
        if self.start_index > 0:
            #  assert self.start_index % self.world_size == 0
            indices = indices[self.start_index :]

        # Slice indices by rank to avoid duplicates.
        indices = indices[self.rank : self.total_size : self.world_size]

        # Separate from data loading workers (which use multiprocessing), we also have the option
        # to use multi-threading (within workers).
        num_threads = self.num_threads

        # Slice the indices by data loader worker rank to avoid duplicates.
        worker_info = torch.utils.data.get_worker_info()
        if worker_info is not None:
            # Note that each data loading worker gathers a whole batch at a time, and the workers
            # are called round-robin by rank. So to slice these up in a way that preserves order, regardless
            # of the number of workers, we should give worker 0 the first chunk of `device_batch_size` indices,
            # worker 1 the 2nd chunk of `device_train_batch_size` indices, etc...
            truncated_size = self.device_batch_size * (len(indices) // self.device_batch_size)
            left_overs = indices[truncated_size + worker_info.id :: worker_info.num_workers]
            indices = (
                indices[:truncated_size]
                .reshape((-1, self.device_batch_size))[worker_info.id :: worker_info.num_workers]  # type: ignore
                .reshape((-1,))
            )
            indices = np.concatenate([indices, left_overs])
        elif num_threads is None:
            # If `num_threads` hasn't been specified and we're not using multiprocessing we'll try to guess
            # a good number of threads.
            num_threads = 4

        # Finally, potentially slice by threads.
        if num_threads:
            # In order to stay ahead of training the total queue size (sum across all threads)
            # should be bigger than the batch size.
            queue_size = math.ceil(self.device_batch_size * 2 / num_threads)

            thread_generators = []
            for i in range(num_threads):
                generator = (self._get_dataset_item(int(idx)) for idx in indices[i::num_threads])
                thread_generators.append(
                    threaded_generator(generator, maxsize=queue_size, thread_name=f"data thread {i}")
                )

            return (x for x in roundrobin(*thread_generators))
        else:
            return (self._get_dataset_item(int(idx)) for idx in indices)

    def _get_dataset_item(self, idx: int) -> Dict[str, Any]:
        item = self.dataset[idx]
        if isinstance(item, dict):
            return dict(**item, index=idx)
        elif dataclasses.is_dataclass(item):
            return dict(**dataclasses.asdict(item), index=idx)  # type: ignore
        else:
            return {"input_ids": item, "index": idx}
def build_train_dataloader(
    train_config: TrainConfig,
    *,
    world_size: Optional[int] = None,
    rank: Optional[int] = None,
    fs_local_rank: Optional[int] = None,
    include_instance_metadata: bool = False,
) -> DataLoader:
    assert train_config.device_train_batch_size is not None
    seed = train_config.data.seed if train_config.data.seed is not None else train_config.seed
    collator = build_collator(train_config)
    if train_config.data.custom_dataset:
        if train_config.data.paths is not None or train_config.data.datasets is not None:
            raise OLMoConfigurationError(
                "custom_dataset_class is mutually exclusive with DataConfig.paths and DataConfig.datasets"
            )

    else:
        dataset = build_memmap_dataset(
            train_config, train_config.data, include_instance_metadata=include_instance_metadata
        )
    work_dir = Path(train_config.save_folder) / "train_data"
    if get_global_rank() == 0:
        if work_dir.is_dir() and not train_config.save_overwrite:

            )
        else:
            work_dir.mkdir(exist_ok=True, parents=True)
    dataset = IterableDataset(
        dataset,  # type: ignore
        train_config.global_train_batch_size,
        seed=seed,
        epoch=train_config.epoch or 0,
        shuffle=True,
        drop_last=train_config.data.drop_last,
        world_size=world_size,
        rank=rank,
        fs_local_rank=fs_local_rank,
        work_dir=work_dir,
    )
    barrier()
    out = DataLoader(
        dataset,
        batch_size=train_config.device_train_batch_size,
        drop_last=train_config.data.drop_last,
        collate_fn=collator,
        num_workers=train_config.data.num_workers,
        pin_memory=train_config.data.pin_memory,
        prefetch_factor=None if train_config.data.num_workers == 0 else train_config.data.prefetch_factor,
        persistent_workers=False if train_config.data.num_workers == 0 else train_config.data.persistent_workers,
        timeout=train_config.data.timeout,
    )
    return out

Объясни подробно как работает данный датасет для обучения LLM 

Отлично, давайте подробно разберём, как устроен этот пайплайн данных для обучения больших языковых моделей (LLM). Он состоит из двух ключевых компонентов, которые работают в связке: `MemMapDataset` и `IterableDataset`.

### Общая идея: Зачем всё так сложно?

Обучение LLM требует обработки **огромных объёмов данных** — терабайты токенизированного текста. Эти данные невозможно целиком загрузить в оперативную память (RAM). Поэтому нужна система, которая решает две основные проблемы:

1.  **Эффективный доступ к данным на диске:** Как читать нужные фрагменты данных с диска, не загружая весь файл в память и делая это очень быстро.
2.  **Эффективная подача данных в модель:** Как организовать перемешивание данных, распределить их по всем видеокартам (GPU) в кластере и обеспечить возможность возобновления обучения с того же места, где оно остановилось.

`MemMapDataset` решает первую проблему, а `IterableDataset` — вторую.

---

### 1. `MemMapDataset`: Умный доступ к данным на диске

Это "фундамент" нашего датасета. Его главная задача — предоставить доступ к данным, которые хранятся на диске в виде бинарных файлов, как если бы они были в памяти.

#### Ключевая технология: Memory Mapping (mmap)

Представьте, что у вас есть огромный файл с токенами на диске (например, 100 ГБ). Вместо того чтобы читать его кусок за куском с помощью `file.read()`, мы используем **memory mapping**. Операционная система "проецирует" этот файл в виртуальное адресное пространство вашего процесса.

*   **Что это даёт?** Вы можете обращаться к этому файлу как к обычному массиву в памяти (например, в `numpy`). Когда вы запрашиваете определённый срез данных (например, с 1,000,000-го по 1,001,024-й токен), операционная система сама подгружает с диска только эту нужную "страницу" данных. Это **невероятно быстро и эффективно**, так как все низкоуровневые операции по чтению с диска берёт на себя ОС, которая делает это оптимально.
*   **Аналогия:** Представьте огромную книгу. Вместо того чтобы переносить всю книгу на свой стол (в RAM), вы получаете "волшебную закладку" (mmap), которая позволяет вам мгновенно открыть и прочитать любую страницу, не двигая саму книгу (файл).

#### Как работает `MemMapDataset`

1.  **Инициализация (`__init__`)**:
    *   Принимает на вход список путей (`paths`) к файлам с токенами. Эти файлы — просто бинарные последовательности чисел (ID токенов), сохранённые в формате `numpy`.
    *   `chunk_size`: Это размер одного примера для модели, обычно равный максимальной длине последовательности (например, 2048 или 4096). Датасет будет "нарезать" большие файлы на куски (чанки) такого размера.
    *   `memmap_dtype`: Тип данных в файле (`np.uint16`, `np.uint32` и т.д.). Важно для правильного расчёта смещений в байтах.
    *   `generate_attention_mask`, `generate_doc_lengths` и т.д.: Флаги для генерации дополнительных данных "на лету" для каждого примера. Например, маска внимания (чтобы модель не обращала внимание на паддинг-токены) или информация о границах документов внутри одного чанка.

2.  **Вычисление смещений (`offsets`)**:
    *   Это очень важный шаг. Класс не читает файлы сразу. Вместо этого он вычисляет, сколько "чанков" (`chunk_size`) помещается в каждый файл.
    *   Свойство `offsets` хранит кортежи `(start_offset, end_offset)` для каждого файла. Например, если у нас два файла:
        *   `file1.mmap` содержит 1000 чанков.
        *   `file2.mmap` содержит 2000 чанков.
    *   `self.offsets` будет выглядеть так: `[(0, 1000), (1000, 3000)]`.
    *   Это позволяет датасету работать с множеством файлов как с одним большим виртуальным датасетом общим размером в 3000 примеров.

3.  **Получение элемента (`__getitem__(index)`)**:
    *   Это сердце класса. Когда вы запрашиваете `dataset[1500]`:
        1.  Код находит, в какой файл попадает этот индекс. В нашем примере `1000 <= 1500 < 3000`, значит, это второй файл (`memmap_index = 1`).
        2.  Вычисляется локальный индекс внутри этого файла: `memmap_local_index = 1500 - 1000 = 500`.
        3.  Вызывается `_read_chunk_from_memmap`. Эта функция **не читает весь файл**. Она вычисляет точное смещение в байтах: `bytes_start = memmap_local_index * item_size * chunk_size`.
        4.  С помощью `get_bytes_range` она запрашивает у файловой системы (локальной или облачной, типа S3) ровно тот байтовый диапазон, который соответствует 500-му чанку.
        5.  Этот маленький бинарный буфер (например, 4096 * 2 байта = 8 КБ) преобразуется в `torch.Tensor`.
    *   Таким образом, для получения одного примера считывается всего несколько килобайт, а не гигабайты.

**Итог по `MemMapDataset`:** это эффективный "ридер" данных с диска, который умеет работать с гигантскими файлами и нарезать их на чанки нужного размера.

### 2. `IterableDataset`: Оркестратор обучения

Теперь, когда у нас есть способ быстро получить *любой* пример по его индексу, нам нужен "дирижёр", который будет управлять процессом подачи данных для распределённого обучения. Эту роль выполняет `IterableDataset`.

Он не хранит данные, а **оборачивает** `MemMapDataset`.

#### Ключевые задачи и их решение

1.  **Детерминированное перемешивание (Shuffle):**
    *   Для хорошего обучения данные нужно перемешивать на каждой эпохе. Но в распределённом режиме важно, чтобы порядок перемешивания был **одинаковым для всех GPU**, иначе обучение будет хаотичным.
    *   **Решение:** Метод `_build_global_indices` создаёт единый для всех "генеральный план" на эпоху. Он берёт все индексы от 0 до `len(dataset) - 1`, перемешивает их с использованием `seed + epoch`, и получает глобальный порядок примеров на эту эпоху.

2.  **Сохранение и переиспользование порядка (`_build_and_save_global_indices`):**
    *   Создание этого глобального списка индексов может занять время, если датасет очень большой.
    *   **Решение:** На первом процессе (`fs_local_rank == 0`) этот перемешанный список индексов сохраняется на диск в файл (`global_indices.npy`). Все остальные процессы просто загружают этот готовый файл. Это ускоряет запуск и гарантирует идентичность порядка.

3.  **Распределение данных по GPU (Distributed Sampling):**
    *   Нужно сделать так, чтобы каждый GPU получал свою уникальную порцию данных, и чтобы за эпоху все данные были обработаны ровно один раз.
    *   **Решение:** В методе `__iter__` используется простая, но гениальная строчка:
        ```python
        indices = indices[self.rank : self.total_size : self.world_size]
        ```
        `rank` — это номер текущего GPU, а `world_size` — общее число GPU. Эта операция нарезки "раздаёт" индексы из глобального списка по GPU. Например, если у нас 4 GPU:
        *   GPU 0 получит индексы 0, 4, 8, 12, ...
        *   GPU 1 получит индексы 1, 5, 9, 13, ...
        *   GPU 2 получит индексы 2, 6, 10, 14, ...
        *   GPU 3 получит индексы 3, 7, 11, 15, ...
        Таким образом, каждый GPU получает уникальный, непересекающийся набор данных.

4.  **Возобновление обучения (Resuming):**
    *   Обучение LLM может прерваться. Нужно уметь возобновлять его с того же самого места.
    *   **Решение:** Параметр `start_index`. Если мы обучились на 1,000,000 примеров и обучение упало, мы можем перезапустить его с `start_index = 1000000`. `IterableDataset` просто отрежет первую часть глобального списка индексов и начнёт с нужного места. Поскольку порядок детерминирован, мы продолжим ровно с той же точки.

5.  **Производительность (Многопоточность):**
    *   GPU потребляет данные очень быстро. Чтобы он не простаивал, данные нужно готовить заранее (pre-fetching).
    *   **Решение:** Внутри `IterableDataset` используется многопоточность (`threaded_generator`). Несколько потоков параллельно запрашивают данные у `MemMapDataset` (через `_get_dataset_item`) и складывают их в очередь. Основной процесс просто забирает уже готовые примеры из этой очереди. Это скрывает задержки чтения с диска.

---

### 3. Сборка всего вместе: `build_train_dataloader`

Эта функция — точка входа. Она связывает всё воедино:

1.  Создаёт `MemMapDataset`, указывая ему пути к данным и параметры модели (размер чанка).
2.  Создаёт `IterableDataset`, оборачивая `MemMapDataset` и передавая ему параметры распределённого обучения (seed, epoch, batch_size).
3.  Наконец, создаёт стандартный `torch.utils.data.DataLoader`, который использует `IterableDataset`. `DataLoader` отвечает за:
    *   **Батчинг:** Группировку отдельных примеров в батчи.
    *   **Многопроцессность (`num_workers`):** Запуск нескольких параллельных процессов для подготовки данных, чтобы ещё больше ускорить загрузку.
    *   **Передачу данных на GPU (`pin_memory`).**

### Итоговая схема работы:

1.  **Настройка:** `build_train_dataloader` создаёт всю цепочку: `DataLoader` -> `IterableDataset` -> `MemMapDataset`.
2.  **Старт эпохи:** `IterableDataset` создаёт или загружает с диска глобальный перемешанный список индексов на текущую эпоху.
3.  **Цикл обучения:**
    *   Основной цикл `for batch in dataloader:` запрашивает новый батч.
    *   `DataLoader` через своих воркеров обращается к `IterableDataset`.
    *   `IterableDataset` выдаёт воркерам их порцию индексов (уже нарезанную по `rank`'ам).
    *   Воркеры в несколько потоков запрашивают у `MemMapDataset` данные по этим индексам.
    *   `MemMapDataset` делает быстрый `mmap`-доступ к диску, читает только нужные килобайты данных для каждого индекса и возвращает `torch.Tensor`.
    *   `DataLoader` собирает тензоры в батч и передаёт его в модель на GPU.

Эта архитектура является очень масштабируемой и производительной, позволяя эффективно обучать модели на петабайтных датасетах, которые во много раз превышают объём доступной оперативной памяти.

- ==объясни всю функцию __iter__==
Конечно! Давайте разберём функцию `__iter__` во всех деталях. Это, пожалуй, самая важная и сложная часть всего класса `IterableDataset`, так как именно здесь происходит вся "магия" распределения данных.

### Общая цель функции `__iter__`

Основная задача этой функции — вернуть **итератор**. Когда вы в цикле обучения пишете `for batch in dataloader:`, именно этот итератор будет вызываться для получения каждого следующего примера данных.

Итератор, который создаёт эта функция, должен быть "умным" и решать следующие задачи:

1.  **Детерминизм:** Гарантировать, что при одинаковых `seed` и `epoch` порядок данных будет абсолютно одинаковым.
2.  **Возобновляемость:** Позволить начать итерацию не с самого начала, а с определённого места (`start_index`).
3.  **Распределённость:** Выдать каждому процессу (GPU) его уникальную, непересекающуюся часть данных.
4.  **Эффективность:** Организовать параллельную загрузку данных, чтобы GPU никогда не простаивал в ожидании.

Теперь пройдёмся по коду шаг за шагом.

---

### Этапы работы (Шаг за шагом)

#### Шаг 1: Получение глобального порядка данных

```python
def __iter__(self) -> Iterator[Dict[str, Any]]:
    indices = self.get_global_indices()
```

*   **Что делает код?** Он вызывает `self.get_global_indices()`. Эта функция либо создаёт с нуля, либо (что более вероятно) загружает с диска заранее созданный `numpy` массив. Этот массив `indices` содержит индексы всех примеров в `MemMapDataset` (от 0 до N-1), перемешанные в детерминированном порядке для текущей эпохи.
*   **Зачем это нужно?** Это наш **"генеральный план"** на эпоху. Все GPU в кластере будут работать с *абсолютно одинаковой* копией этого `indices`. Это критически важно для координации. Если бы у каждого GPU был свой случайный порядок, мы бы не смогли правильно разделить данные.

#### Шаг 2: Ограничение максимального количества примеров

```python
    # Truncate to max_examples.
    if self.max_examples is not None:
        assert self.max_examples % self.world_size == 0
        indices = indices[: self.max_examples]
```

*   **Что делает код?** Если в конфигурации задан параметр `max_examples` (например, для короткого тестового прогона), он "обрезает" наш "генеральный план", оставляя только первые `max_examples` индексов.
*   **Зачем это нужно?** Позволяет гибко управлять длиной эпохи. Например, можно сказать "пройти не весь датасет, а только первые 10 миллионов примеров". `assert` проверяет, что это число делится на количество GPU, чтобы каждый получил равное количество работы.

#### Шаг 3: Возобновление с контрольной точки

```python
    # Start at the specified index.
    if self.start_index > 0:
        #  assert self.start_index % self.world_size == 0
        indices = indices[self.start_index :]
```

*   **Что делает код?** Если обучение было прервано и мы возобновляем его, `start_index` будет больше нуля. Этот код отбрасывает уже обработанные индексы из начала списка.
*   **Зачем это нужно?** Это **ключевой механизм возобновления обучения**. Если мы остановились после обработки 1,000,000 примеров, мы просто передаём `start_index=1000000`, и итерация начнётся ровно с того места, где мы остановились, не пропустив и не повторив ни одного примера.

#### Шаг 4: Распределение данных по GPU (самый важный шаг)

```python
    # Slice indices by rank to avoid duplicates.
    indices = indices[self.rank : self.total_size : self.world_size]
```

*   **Что делает код?** Это элегантный способ нарезки массива в Python. Он выбирает элементы из `indices`, начиная с индекса `self.rank` с шагом `self.world_size`.
*   **Зачем это нужно?** Это и есть **распределённая выборка**. Представьте `world_size = 4` GPU.
    *   GPU 0 (`rank=0`) получит индексы: 0, 4, 8, 12, ...
    *   GPU 1 (`rank=1`) получит индексы: 1, 5, 9, 13, ...
    *   GPU 2 (`rank=2`) получит индексы: 2, 6, 10, 14, ...
    *   GPU 3 (`rank=3`) получит индексы: 3, 7, 11, 15, ...
    Таким образом, общий "генеральный план" раздаётся GPU как колода карт, по одной карте каждому по очереди. Каждый получает свою уникальную порцию.

#### Шаг 5: Распределение данных по воркерам `DataLoader`'а

```python
    worker_info = torch.utils.data.get_worker_info()
    if worker_info is not None:
        # ... сложная логика нарезки ...
```

*   **Что делает код?** Теперь у нас есть список индексов для *одного GPU*. Но `DataLoader` может использовать несколько CPU-процессов (`num_workers`) для загрузки данных для этого GPU. Этот блок кода делит список индексов этого GPU между его воркерами.
*   **Зачем это нужно?** Это второй уровень параллелизма. Вместо того чтобы один CPU-процесс готовил все данные для GPU, мы можем задействовать, например, 8 процессов. Логика здесь сделана так, чтобы каждый воркер получал целые "микро-батчи" (`device_batch_size`), что может быть эффективнее.

#### Шаг 6: Распределение данных по потокам (опционально)

```python
    if num_threads:
        # ... создание нескольких threaded_generator ...
        return (x for x in roundrobin(*thread_generators))
    else:
        return (self._get_dataset_item(int(idx)) for idx in indices)
```

*   **Что делает код?** Это третий, самый низкий уровень параллелизма. Даже внутри одного CPU-воркера мы можем создать несколько потоков (`threads`) для чтения данных.
*   **Зачем это нужно?** Это **оптимизация производительности**. Чтение с диска (даже быстрого) — это операция ввода-вывода (I/O). Пока один поток ждёт ответа от диска, другой может обрабатывать уже полученные данные или делать новый запрос.
    *   **Если `num_threads` > 0:** Создаётся несколько генераторов, каждый работает в своём потоке и кладёт готовые примеры в очередь (`threaded_generator`). `roundrobin` затем поочерёдно забирает готовые примеры из всех очередей, обеспечивая непрерывный поток данных.
    *   **Если `num_threads` не задан:** Возвращается простой генератор, который последовательно, один за другим, запрашивает примеры по индексам.

#### Финальный шаг: Получение самого элемента

Внутри генератора вызывается `self._get_dataset_item(int(idx))`. Эта функция, в свою очередь, делает `self.dataset[idx]`, что вызывает `__getitem__` нашего `MemMapDataset`. И именно в этот момент происходит быстрое чтение нужного чанка данных с диска.

### Итоговая иерархия распределения данных

Можно представить себе воронку:

1.  **Глобальный уровень:** `indices` (миллиарды примеров, единый для всех).
2.  **Уровень GPU:** `indices[rank::world_size]` (сотни миллионов примеров для одного GPU).
3.  **Уровень CPU-воркера:** `indices` для GPU делятся между `num_workers` (десятки миллионов для одного воркера).
4.  **Уровень потока:** `indices` для воркера делятся между `num_threads` (миллионы для одного потока).
5.  **Исполнение:** Поток берёт один `idx`, передаёт его в `MemMapDataset`, который читает с диска ~8 КБ данных.

Таким образом, функция `__iter__` — это сложный, многоуровневый оркестратор, который гарантирует, что правильные данные, в правильном порядке, максимально эффективно попадут на каждый GPU в кластере.
